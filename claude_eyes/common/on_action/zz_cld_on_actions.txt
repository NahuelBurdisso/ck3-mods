# Claude Eyes - the watchdog.
#
# RULE, and the whole reason this file looks the way it does: an on_action can
# have only ONE `trigger` block across the entire playset, and `effect` blocks
# are resolved by file load order rather than merged. So this file NEVER adds a
# `trigger` or an `effect` to a shared vanilla/AGOT on_action. It only appends
# to the `on_actions = { }` list, which IS additive, and every gate lives inside
# our own nested on_action. Grand Remembrance documents the same rule at
# common/on_action/zz_gr_memory_tracking.txt:11-19, where breaking it silently
# gated vanilla's own title handling to players only.
#
# Named zz_ so it loads after vanilla and AGOT. That only affects which `effect`
# block wins, which is exactly what we are staying out of.
#
# Scope notes come from docs/api/1.19.0.6/on_actions.log plus the vanilla
# on_action files' own header comments.

# =========================================================================
# SUCCESSION AND DEATH
# =========================================================================
# root = the dying character
on_death = {
	on_actions = { cld_on_death }
}

cld_on_death = {
	effect = {
		if = {
			limit = { is_ai = no }
			debug_log = "CLD;TRIG;kind=death;who=player"
			cld_digest_effect = yes
		}
	}
}

# =========================================================================
# WAR
# =========================================================================
# root = none. The available scopes are the casus belli's on_declaration set,
# so every read below is guarded with `exists` before it is dereferenced.
on_war_started = {
	on_actions = { cld_on_war_started }
}

cld_on_war_started = {
	effect = {
		if = {
			limit = {
				exists = scope:attacker
				scope:attacker = { is_ai = no }
			}
			debug_log = "CLD;TRIG;kind=war_started;side=attacker"
			scope:attacker = { cld_digest_effect = yes }
		}
		if = {
			limit = {
				exists = scope:defender
				scope:defender = { is_ai = no }
			}
			debug_log = "CLD;TRIG;kind=war_started;side=defender"
			scope:defender = { cld_digest_effect = yes }
		}
	}
}

# =========================================================================
# THE PERIODIC HEARTBEAT
# root = character, fires four times a year. This is the cheapest hook the
# engine exposes with a character root, and it is what carries the situations
# that have NO on_action at all - factions forming, truces expiring, the heir
# changing, health decaying, an epidemic arriving. None of those are detected
# in script: the digest is emitted unconditionally and the reader diffs two
# consecutive digests to find the crossing. Keeping the comparison outside the
# game is what lets a new watch condition be added without another 7-minute
# restart.
# =========================================================================
quarterly_playable_pulse = {
	on_actions = { cld_quarterly }
}

cld_quarterly = {
	effect = {
		if = {
			limit = { is_ai = no }
			debug_log = "CLD;TRIG;kind=heartbeat;who=player"
			cld_digest_effect = yes
		}
	}
}
