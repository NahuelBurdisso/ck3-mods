# AGOT: War of the Small Folk - secret types
#
# Funding a brotherhood is a real CK3 secret, not a character flag. That buys
# the whole vanilla apparatus for free: the spymaster can find it, a rival can
# blackmail you with it, and exposure runs through machinery that already
# works. A flag would have bought none of it.
#
# Schema copied from secret_embezzler (00_secret_types.txt:619).
#
# scope:secret_owner  = the patron
# scope:secret_target = the ruler whose county is burning

wotsf_secret_funding_outlaws = {
	category = civil

	is_valid = {
		scope:secret_owner = { is_alive = yes }
	}

	# Shunned, not criminal. No realm has a law against paying men you never met,
	# but every lord in the room knows what it means.
	is_shunned = {
		always = yes
	}

	is_criminal = {
		always = no
	}

	on_expose = {
		save_scope_as = secret
		scope:secret_owner = {
			trigger_event = { id = wotsf.0304 days = 1 }
		}
	}
}
