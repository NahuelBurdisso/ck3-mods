#   band 2  50-74  wotsf_suffering_2_modifier   "Tierra doliente"
#   band 3  75-100 wotsf_suffering_3_modifier   "Tierra quemada"
#

############################################################################
# PHASE 2 - CHEVAUCHEE
#
# The chevauchee itself is county state, not army state: add_army_modifier
# does not exist in 1.19. The county modifier is the visible marker, and
# wotsf_chevauchee_months is what the monthly pulse counts down.
#
# Every modifier here is applied with an explicit duration, so none of them
# can leak if a variable is ever lost.
############################################################################

wotsf_chevauchee_active_modifier = {
	icon = county_modifier_control_negative
	development_growth_factor = -0.5
	county_opinion_add = -20
}

# THIS MODIFIER *IS* THE 6-MONTH COOLDOWN. Its presence is the entire check in
# wotsf_county_chevauchee_ready_trigger, and its duration is the only clock -
# nothing counts it down, the engine expires it. No stat effects on purpose: it
# is a gate and a piece of information, not a penalty, and the suffering bands
# are already doing the hurting. Icon-only modifiers are a normal vanilla shape
# (00_county_modifiers.txt: reclaim_britannia_culture_modifier).
wotsf_chevauchee_scorched_modifier = {
	icon = county_modifier_development_negative
}

# On the commander. They are busy burning, not marching.
wotsf_chevauchee_commander_modifier = {
	icon = travel_negative
	movement_speed = -0.25
}

# On the perpetrator. Refreshed, not stacked, by every new chevauchee.
wotsf_chevauchee_clergy_modifier = {
	icon = piety_negative
	clergy_opinion = -10
}

# The defender chose to garrison the county (design 3.4a, redesigned:
# there is no effect in 1.19 that removes men from a raised army).
wotsf_defended_modifier = {
	icon = county_modifier_control_positive
	county_opinion_add = 15
	monthly_county_control_growth_add = 0.1
}

wotsf_troops_diverted_modifier = {
	icon = martial_negative
	levy_size = -0.15
	levy_reinforcement_rate = -0.25
}

# "Son of the Scourge". Applied to the children of a wotsf_scourge holder,
# stripped from all of them when the parent dies.
wotsf_scourge_child_modifier = {
	icon = diplomacy_negative
	general_opinion = -5
}

############################################################################
# PHASE 3 - BROTHERHOODS
#
# Modifier values are static, so wotsf_brotherhood_strength cannot drive one
# modifier's numbers. Two tiers instead, exactly one at a time, siblings
# stripped before the new one is added - the same shape as the suffering
# bands, and for the same reason: two different county modifiers STACK.
#
# THESE TWO CARRY NO DURATION, unlike every Phase 2 modifier. The band is
# alive until something kills it, so there is no honest number to put here.
# That makes wotsf_end_brotherhood_effect the only thing that removes them,
# and every exit path has to go through it or the modifier is permanent.
# See R16 in the Phase 3 plan.
############################################################################

wotsf_brotherhood_modifier = {
	icon = county_modifier_control_negative
	tax_mult = -0.2
	monthly_county_control_growth_add = -0.25
	county_opinion_add = -15
}
